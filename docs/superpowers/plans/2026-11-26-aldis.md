# Aldis Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Personal Telegram bot that answers news/web questions through the owner's ChatGPT subscription (`codex exec`) with Exa search, replying in short sequential messages.

**Architecture:** Bun + grammy long-polling bot. Each user turn spawns one `codex exec --json` process (shell disabled, read-only sandbox, Exa hosted MCP), streams its JSONL events, shows tool progress in one edited status message, then sends the final answer as paragraph chunks converted to Telegram HTML. Chat → Codex thread id mapping in `bun:sqlite`.

**Tech Stack:** Bun 1.3, TypeScript (strict), grammy, @grammyjs/auto-retry, bun:sqlite, bun:test, codex-cli ≥ 0.172.

**Spec:** `docs/superpowers/specs/2026-11-26-onlytext-design.md`

## Global Constraints

- Name constant: `APP_NAME = "Aldis"` in `src/config.ts`; nowhere else hard-codes the name except deploy files/README.
- One file = one responsibility; no file over ~150 lines.
- Codex flags (both new and resume runs): `--json --skip-git-repo-check --ignore-user-config -c sandbox_mode="read-only" --disable shell_tool --disable unified_exec --disable apps --disable plugins --disable browser_use --disable in_app_browser --disable computer_use --disable view_image -c mcp_servers.exa.url="…"`. `codex exec resume` does NOT accept `-s`, hence `-c sandbox_mode`.
- Child env: only `PATH`, `HOME`, `LANG`, `LC_ALL`, `TZ`, plus `CODEX_HOME`.
- Final answer = text of the LAST `agent_message` item.
- Defaults: `TURN_TIMEOUT_SEC=180`, `IDLE_RESET_HOURS=6`, `SESSION_RETENTION_DAYS=7`, `BRIEF_TOPICS=world,turkey,tech,markets`, `CHUNK_MAX=1200`, `CHUNK_DELAY_MS=900`, `TZ_NAME=Europe/Istanbul`, `DATA_DIR=./data`.
- Link previews disabled on every bot message.
- User-facing copy in Turkish, in `src/bot/texts.ts` only.
- Commit after every task; message ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Do not touch `assets/logo/` (logo is produced by a separate agent).

import { join, resolve } from "node:path";

export const APP_NAME = "Aldis";

export interface Config {
  telegramToken: string;
  allowedUserId: number;
  exaApiKey: string;
  dataDir: string;
  codexBin: string;
  codexHome: string;
  codexModel?: string;
  turnTimeoutSec: number;
  idleResetHours: number;
  sessionRetentionDays: number;
  briefTopics: string[];
  chunkMax: number;
  chunkDelayMs: number;
  tzName: string;
}

type Env = Record<string, string | undefined>;

function str(env: Env, key: string): string | undefined {
  const v = env[key]?.trim();
  return v ? v : undefined;
}

function required(env: Env, key: string): string {
  const v = str(env, key);
  if (!v) throw new Error(`Missing required env var ${key}`);
  return v;
}

function positiveInt(env: Env, key: string, fallback: number): number {
  const raw = str(env, key);
  if (raw === undefined) return fallback;
  const n = Number(raw);
  if (!Number.isInteger(n) || n <= 0) {
    throw new Error(`Env var ${key} must be a positive integer, got "${raw}"`);
  }
  return n;
}

export function loadConfig(env: Env = process.env): Config {
  const telegramToken = required(env, "TELEGRAM_BOT_TOKEN");
  const allowedRaw = required(env, "ALLOWED_USER_ID");
  const allowedUserId = Number(allowedRaw);
  if (!Number.isInteger(allowedUserId)) {
    throw new Error(`Env var ALLOWED_USER_ID must be an integer, got "${allowedRaw}"`);
  }
  const dataDir = resolve(str(env, "DATA_DIR") ?? "./data");
  return {
    telegramToken,
    allowedUserId,
    exaApiKey: str(env, "EXA_API_KEY") ?? "",
    dataDir,
    codexBin: str(env, "CODEX_BIN") ?? "codex",
    codexHome: resolve(str(env, "CODEX_HOME") ?? join(dataDir, "codex")),
    codexModel: str(env, "CODEX_MODEL"),
    turnTimeoutSec: positiveInt(env, "TURN_TIMEOUT_SEC", 180),
    idleResetHours: positiveInt(env, "IDLE_RESET_HOURS", 6),
    sessionRetentionDays: positiveInt(env, "SESSION_RETENTION_DAYS", 7),
    briefTopics: (str(env, "BRIEF_TOPICS") ?? "world,turkey,tech,markets")
      .split(",")
      .map((t) => t.trim())
      .filter(Boolean),
    chunkMax: positiveInt(env, "CHUNK_MAX", 1200),
    chunkDelayMs: positiveInt(env, "CHUNK_DELAY_MS", 900),
    tzName: str(env, "TZ_NAME") ?? "Europe/Istanbul",
  };
}

type Level = "debug" | "info" | "warn" | "error";

const ORDER: Record<Level, number> = { debug: 10, info: 20, warn: 30, error: 40 };
const minLevel = ORDER[(process.env.LOG_LEVEL as Level) ?? "info"] ?? ORDER.info;

function fmt(arg: unknown): string {
  if (arg instanceof Error) return arg.stack ?? arg.message;
  if (typeof arg === "string") return arg;
  try {
    return JSON.stringify(arg);
  } catch {
    return String(arg);
  }
}

function emit(level: Level, args: unknown[]): void {
  if (ORDER[level] < minLevel) return;
  const line = `[${level}] ${args.map(fmt).join(" ")}`;
  if (level === "warn" || level === "error") console.error(line);
  else console.log(line);
}

export const log = {
  debug: (...args: unknown[]) => emit("debug", args),
  info: (...args: unknown[]) => emit("info", args),
  warn: (...args: unknown[]) => emit("warn", args),
  error: (...args: unknown[]) => emit("error", args),
};
```

```ts
export interface ChunkOptions {
  max: number;
  min?: number;
}

function splitBlocks(text: string): string[] {
  const blocks: string[] = [];
  let current: string[] = [];
  let inFence = false;
  const flush = () => {
    const block = current.join("\n").trim();
    if (block) blocks.push(block);
    current = [];
  };
  for (const line of text.replace(/\r\n/g, "\n").split("\n")) {
    if (line.trimStart().startsWith("```")) inFence = !inFence;
    if (!inFence && line.trim() === "") {
      flush();
      continue;
    }
    current.push(line);
  }
  flush();
  return blocks;
}

function splitHard(s: string, max: number): string[] {
  const out: string[] = [];
  for (let i = 0; i < s.length; i += max) out.push(s.slice(i, i + max));
  return out;
}

function pack(pieces: string[], sep: string, max: number): string[] {
  const out: string[] = [];
  let cur = "";
  for (const piece of pieces) {
    if (!cur) cur = piece;
    else if (cur.length + sep.length + piece.length <= max) cur += sep + piece;
    else {
      out.push(cur);
      cur = piece;
    }
  }
  if (cur) out.push(cur);
  return out;
}

function splitOversize(block: string, max: number): string[] {
  if (block.length <= max) return [block];
  const lines = block.split("\n").flatMap((line) => {
    if (line.length <= max) return [line];
    const sentences = line
      .split(/(?<=[.!?…])\s+/)
      .flatMap((s) => (s.length <= max ? [s] : splitHard(s, max)));
    return pack(sentences, " ", max);
  });
  return pack(lines, "\n", max);
}

export function chunkText(text: string, { max, min = 280 }: ChunkOptions): string[] {
  const blocks = splitBlocks(text).flatMap((b) => splitOversize(b, max));
  const out: string[] = [];
  let cur = "";
  for (const block of blocks) {
    if (!cur) cur = block;
    else if (cur.length < min && cur.length + 2 + block.length <= max) cur += `\n\n${block}`;
    else {
      out.push(cur);
      cur = block;
    }
  }
  if (cur) out.push(cur);
  return out;
}
```

```ts
export function escapeHtml(s: string): string {
  return s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
}

function escapeAttr(s: string): string {
  return escapeHtml(s).replace(/"/g, "&quot;");
}

function formatInline(text: string): string {
  const slots: string[] = [];
  const stash = (html: string) => `\u0000${slots.push(html) - 1}\u0000`;
  let s = text.replace(/`([^`]+)`/g, (_, code: string) => stash(`<code>${escapeHtml(code)}</code>`));
  s = s.replace(/\[([^\]]+)\]\((https?:\/\/[^\s)]+)\)/g, (_, label: string, url: string) =>
    stash(`<a href="${escapeAttr(url)}">${escapeHtml(label)}</a>`),
  );
  s = escapeHtml(s);
  s = s.replace(/\*\*(.+?)\*\*/g, "<b>$1</b>").replace(/__(.+?)__/g, "<b>$1</b>");
  s = s.replace(/(^|[^\w*])\*(?!\s)(.+?)(?<!\s)\*(?!\w)/g, "$1<i>$2</i>");
  s = s.replace(/(^|[^\w])_(?!\s)(.+?)(?<!\s)_(?!\w)/g, "$1<i>$2</i>");
  return s.replace(/\u0000(\d+)\u0000/g, (_, i: string) => slots[Number(i)] ?? "");
}

function formatLine(line: string): string {
  const heading = line.match(/^\s{0,3}#{1,6}\s+(.*)$/);
  if (heading) return `<b>${formatInline(heading[1] ?? "")}</b>`;
  const bullet = line.match(/^(\s*)[-*+]\s+(.*)$/);
  if (bullet) return `${bullet[1] ?? ""}• ${formatInline(bullet[2] ?? "")}`;
  return formatInline(line);
}

export function markdownToTelegramHtml(md: string): string {
  const out: string[] = [];
  let fence: string[] | null = null;
  for (const line of md.replace(/\r\n/g, "\n").split("\n")) {
    if (line.trimStart().startsWith("```")) {
      if (fence) {
        out.push(`<pre>${escapeHtml(fence.join("\n"))}</pre>`);
        fence = null;
      } else {
        fence = [];
      }
      continue;
    }
    if (fence) fence.push(line);
    else out.push(formatLine(line));
  }
  if (fence) out.push(`<pre>${escapeHtml(fence.join("\n"))}</pre>`);
  return out.join("\n");
}
```

import type { BrainEvent } from "./types";

type Json = Record<string, unknown>;

function isObj(v: unknown): v is Json {
  return typeof v === "object" && v !== null;
}

function toolDetail(args: unknown): string | undefined {
  if (!isObj(args)) return undefined;
  const v = args.query ?? args.url ?? args.urls;
  if (typeof v === "string") return v;
  if (Array.isArray(v) && typeof v[0] === "string") return v[0];
  return undefined;
}

export function parseCodexLine(line: string): BrainEvent | null {
  const trimmed = line.trim();
  if (!trimmed.startsWith("{")) return null;
  let ev: unknown;
  try {
    ev = JSON.parse(trimmed);
  } catch {
    return null;
  }
  if (!isObj(ev)) return null;
  const item = isObj(ev.item) ? ev.item : undefined;
  switch (ev.type) {
    case "thread.started":
      return typeof ev.thread_id === "string" ? { type: "session", sessionId: ev.thread_id } : null;
    case "item.started":
      if (item?.type === "mcp_tool_call" && typeof item.tool === "string") {
        return { type: "tool_start", tool: item.tool, detail: toolDetail(item.arguments) };
      }
      return null;
    case "item.completed":
      if (item?.type === "agent_message" && typeof item.text === "string") return { type: "message", text: item.text };
      if (item?.type === "error" && typeof item.message === "string") return { type: "warning", message: item.message };
      return null;
    case "turn.completed":
      return { type: "turn_done" };
    case "turn.failed": {
      const err = isObj(ev.error) ? ev.error : {};
      return { type: "error", message: typeof err.message === "string" ? err.message : "turn failed" };
    }
    case "error":
      return { type: "error", message: typeof ev.message === "string" ? ev.message : "unknown error" };
    default:
      return null;
  }
}

```ts
import { buildCodexArgs } from "./codexArgs";
import { parseCodexLine } from "./codexEvents";
import type { Brain, BrainEvent, BrainRunOptions } from "./types";

export interface CodexBrainOptions {
  bin: string;
  codexHome: string;
  workDir: string;
  exaApiKey: string;
  model?: string;
}

const SAFE_ENV = ["PATH", "HOME", "LANG", "LC_ALL", "TZ"] as const;

export function childEnv(codexHome: string, base: Record<string, string | undefined> = process.env): Record<string, string> {
  const env: Record<string, string> = { CODEX_HOME: codexHome };
  for (const key of SAFE_ENV) {
    const v = base[key];
    if (v) env[key] = v;
  }
  return env;
}

async function* readLines(stream: ReadableStream<Uint8Array>): AsyncIterable<string> {
  const decoder = new TextDecoder();
  let buf = "";
  for await (const chunk of stream) {
    buf += decoder.decode(chunk, { stream: true });
    let nl: number;
    while ((nl = buf.indexOf("\n")) >= 0) {
      yield buf.slice(0, nl);
      buf = buf.slice(nl + 1);
    }
  }
  buf += decoder.decode();
  if (buf) yield buf;
}

function tail(text: string, lines: number): string {
  return text.trim().split("\n").slice(-lines).join("\n");
}

export class CodexBrain implements Brain {
  constructor(private readonly opts: CodexBrainOptions) {}

  async *run({ prompt, sessionId, signal }: BrainRunOptions): AsyncIterable<BrainEvent> {
    const args = buildCodexArgs({ sessionId, exaApiKey: this.opts.exaApiKey, model: this.opts.model });
    const proc = Bun.spawn([this.opts.bin, ...args], {
      cwd: this.opts.workDir,
      env: childEnv(this.opts.codexHome),
      stdin: "pipe",
      stdout: "pipe",
      stderr: "pipe",
    });
    const onAbort = () => proc.kill("SIGTERM");
    signal?.addEventListener("abort", onAbort, { once: true });
    const stderr = new Response(proc.stderr).text();
    proc.stdin.write(prompt);
    proc.stdin.end();

    let sawDone = false;
    let sawError = false;
    try {
      for await (const line of readLines(proc.stdout)) {
        const ev = parseCodexLine(line);
        if (!ev) continue;
        if (ev.type === "turn_done") sawDone = true;
        if (ev.type === "error") sawError = true;
        yield ev;
      }
      const code = await proc.exited;
      if (signal?.aborted) {
        yield { type: "error", message: "aborted" };
      } else if (code !== 0 && !sawError) {
        yield { type: "error", message: tail(await stderr, 5) || `codex exited with code ${code}` };
      } else if (!sawDone && !sawError) {
        yield { type: "error", message: "codex ended without completing the turn" };
      }
    } finally {
      signal?.removeEventListener("abort", onAbort);
      if (proc.exitCode === null) proc.kill("SIGKILL");
    }
  }
}

export interface SessionRecord {
  chatId: number;
  sessionId: string;
  updatedAt: number;
}

interface Row {
  chat_id: number;
  session_id: string;
  updated_at: number;
}

export class SessionStore {
  constructor(private readonly db: Database) {}

  get(chatId: number): SessionRecord | null {
    const row = this.db
      .query("SELECT chat_id, session_id, updated_at FROM sessions WHERE chat_id = ?")
      .get(chatId) as Row | null;
    return row ? { chatId: row.chat_id, sessionId: row.session_id, updatedAt: row.updated_at } : null;
  }

  set(chatId: number, sessionId: string, now: number = Date.now()): void {
    this.db
      .query(
        "INSERT INTO sessions (chat_id, session_id, updated_at) VALUES (?, ?, ?) " +
          "ON CONFLICT(chat_id) DO UPDATE SET session_id = excluded.session_id, updated_at = excluded.updated_at",
      )
      .run(chatId, sessionId, now);
  }

  clear(chatId: number): void {
    this.db.query("DELETE FROM sessions WHERE chat_id = ?").run(chatId);
  }
}

export class ChatLock {
  private readonly busy = new Set<number>();

  tryAcquire(chatId: number): (() => void) | null {
    if (this.busy.has(chatId)) return null;
    this.busy.add(chatId);
    let released = false;
    return () => {
      if (released) return;
      released = true;
      this.busy.delete(chatId);
    };
  }

  isBusy(chatId: number): boolean {
    return this.busy.has(chatId);
  }
}

export interface SendChunksOptions {
  delayMs: number;
  sleep?: (ms: number) => Promise<void>;
}

const defaultSleep = (ms: number) => new Promise<void>((r) => setTimeout(r, ms));

function isParseError(err: unknown): boolean {
  return /parse entities|start tag|end tag/i.test(err instanceof Error ? err.message : String(err));
}

export async function sendFormatted(tg: TelegramLike, chatId: number, markdown: string): Promise<void> {
  try {
    await tg.sendMessage(chatId, markdownToTelegramHtml(markdown), { ...NO_PREVIEW, parse_mode: "HTML" });
  } catch (err) {
    if (!isParseError(err)) throw err;
    log.warn("HTML rejected by Telegram, sending plain text", err);
    await tg.sendMessage(chatId, markdown, NO_PREVIEW);
  }
}

export async function sendChunks(
  tg: TelegramLike,
  chatId: number,
  chunks: string[],
  { delayMs, sleep = defaultSleep }: SendChunksOptions,
): Promise<void> {
  for (const [i, chunk] of chunks.entries()) {
    if (i > 0) {
      await tg.sendChatAction(chatId, "typing").catch(() => {});
      await sleep(delayMs);
    }
    await sendFormatted(tg, chatId, chunk);
  }
}

import { log } from "../log";
import { NO_PREVIEW, type TelegramLike } from "./telegram";

export interface StatusOptions {
  minIntervalMs?: number;
  now?: () => number;
}

function truncate(s: string, max: number): string {
  return s.length > max ? `${s.slice(0, max - 1)}…` : s;
}

export function describeTool(tool: string, detail?: string): string {
  const suffix = detail ? `: ${truncate(detail, 70)}` : "";
  if (tool.includes("search")) return `🔎 Arıyorum${suffix}`;
  if (tool.includes("fetch") || tool.includes("crawl")) return `📄 Okuyorum${suffix}`;
  return `🛠 ${tool}${suffix}`;
}

export class StatusMessage {
  private messageId: number | null = null;
  private lastText = "";
  private lastEditAt = 0;
  private readonly minIntervalMs: number;
  private readonly now: () => number;

  constructor(
    private readonly tg: TelegramLike,
    private readonly chatId: number,
    opts: StatusOptions = {},
  ) {
    this.minIntervalMs = opts.minIntervalMs ?? 1500;
    this.now = opts.now ?? Date.now;
  }

  async show(text: string): Promise<void> {
    if (text === this.lastText) return;
    const now = this.now();
    try {
      if (this.messageId === null) {
        const msg = await this.tg.sendMessage(this.chatId, text, NO_PREVIEW);
        this.messageId = msg.message_id;
      } else {
        if (now - this.lastEditAt < this.minIntervalMs) return;
        await this.tg.editMessageText(this.chatId, this.messageId, text, NO_PREVIEW);
      }
      this.lastText = text;
      this.lastEditAt = now;
    } catch (err) {
      log.warn("status update failed", err);
    }
  }

  async clear(): Promise<void> {
    if (this.messageId === null) return;
    const id = this.messageId;
    this.messageId = null;
    await this.tg.deleteMessage(this.chatId, id).catch((err) => log.warn("status delete failed", err));
  }
}

```ts
import { buildTurnPrompt } from "../brain/prompts";
import type { Brain } from "../brain/types";
import { APP_NAME, type Config } from "../config";
import { chunkText } from "../delivery/chunker";
import { sendChunks, sendFormatted } from "../delivery/sender";
import { StatusMessage, describeTool } from "../delivery/status";
import type { TelegramLike } from "../delivery/telegram";
import { startTyping } from "../delivery/typing";
import { log } from "../log";
import type { SessionRecord, SessionStore } from "../store/sessions";
import type { ChatLock } from "./chatLock";
import { describeError, texts } from "./texts";

export interface ConversationDeps {
  brain: Brain;
  sessions: SessionStore;
  lock: ChatLock;
  tg: TelegramLike;
  config: Pick<Config, "turnTimeoutSec" | "idleResetHours" | "chunkMax" | "chunkDelayMs" | "tzName">;
  now?: () => Date;
  sleep?: (ms: number) => Promise<void>;
}

export interface TurnOptions {
  fresh?: boolean;
}

interface TurnOutcome {
  sessionId?: string;
  answer: string;
  error: string | null;
}

const BAD_SESSION = /(session|thread|rollout).*(not found|no such|invalid|missing)/i;

export class Conversation {
  lastTurn: { at: Date; durationMs: number; ok: boolean } | null = null;
  private readonly now: () => Date;

  constructor(private readonly deps: ConversationDeps) {
    this.now = deps.now ?? (() => new Date());
  }

  async runTurn(chatId: number, userText: string, opts: TurnOptions = {}): Promise<void> {
    const { tg, lock } = this.deps;
    const release = lock.tryAcquire(chatId);
    if (!release) {
      await sendFormatted(tg, chatId, texts.busy).catch((err) => log.warn("busy reply failed", err));
      return;
    }
    const startedAt = this.now();
    let ok = false;
    try {
      ok = await this.execute(chatId, userText, opts, startedAt);
    } catch (err) {
      log.error("turn crashed", err);
      await sendFormatted(tg, chatId, texts.crash).catch(() => {});
    } finally {
      release();
      this.lastTurn = { at: startedAt, durationMs: this.now().getTime() - startedAt.getTime(), ok };
    }
  }

  private activeSession(chatId: number, at: Date): SessionRecord | null {
    const rec = this.deps.sessions.get(chatId);
    if (!rec) return null;
    if (at.getTime() - rec.updatedAt > this.deps.config.idleResetHours * 3_600_000) {
      this.deps.sessions.clear(chatId);
      return null;
    }
    return rec;
  }

  private async execute(chatId: number, userText: string, opts: TurnOptions, startedAt: Date): Promise<boolean> {
    const { tg, sessions, config } = this.deps;
    const existing = opts.fresh ? null : this.activeSession(chatId, startedAt);
    const prompt = buildTurnPrompt(userText, { now: startedAt, tzName: config.tzName, appName: APP_NAME }, !existing);

    const status = new StatusMessage(tg, chatId);
    const typing = startTyping(tg, chatId);
    const abort = new AbortController();
    const timer = setTimeout(() => abort.abort(), config.turnTimeoutSec * 1000);
    let outcome: TurnOutcome;
    try {
      outcome = await this.consume(prompt, existing?.sessionId, abort.signal, status);
    } finally {
      clearTimeout(timer);
      typing.stop();
      await status.clear();
    }

    if (outcome.sessionId) sessions.set(chatId, outcome.sessionId, this.now().getTime());
    if (abort.signal.aborted) {
      await sendFormatted(tg, chatId, texts.timeout);
      return false;
    }
    if (outcome.error && !outcome.answer.trim()) {
      if (existing && BAD_SESSION.test(outcome.error)) sessions.clear(chatId);
      log.warn("brain error", outcome.error);
      await sendFormatted(tg, chatId, describeError(outcome.error));
      return false;
    }
    if (!outcome.answer.trim()) {
      await sendFormatted(tg, chatId, texts.empty);
      return false;
    }
    const chunks = chunkText(outcome.answer, { max: config.chunkMax });
    await sendChunks(tg, chatId, chunks, { delayMs: config.chunkDelayMs, sleep: this.deps.sleep });
    return true;
  }

  private async consume(
    prompt: string,
    sessionId: string | undefined,
    signal: AbortSignal,
    status: StatusMessage,
  ): Promise<TurnOutcome> {
    const outcome: TurnOutcome = { sessionId, answer: "", error: null };
    for await (const ev of this.deps.brain.run({ prompt, sessionId, signal })) {
      switch (ev.type) {
        case "session":
          outcome.sessionId = ev.sessionId;
          break;
        case "tool_start":
          await status.show(describeTool(ev.tool, ev.detail));
          break;
        case "message":
          outcome.answer = ev.text;
          break;
        case "warning":
          log.debug("codex warning", ev.message);
          break;
        case "error":
          outcome.error = ev.message;
          break;
        case "turn_done":
          break;
      }
    }
    return outcome;
  }
}

```ts
import type { Dirent } from "node:fs";
import { readdir, rm, rmdir, stat } from "node:fs/promises";
import { join } from "node:path";

export async function pruneOldSessions(root: string, maxAgeDays: number, now: number = Date.now()): Promise<number> {
  const cutoff = now - maxAgeDays * 86_400_000;
  let removed = 0;

  async function walk(dir: string): Promise<boolean> {
    let entries: Dirent[];
    try {
      entries = await readdir(dir, { withFileTypes: true });
    } catch {
      return false;
    }
    let remaining = entries.length;
    for (const entry of entries) {
      const path = join(dir, entry.name);
      if (entry.isDirectory()) {
        if (await walk(path)) {
          await rmdir(path).catch(() => {});
          remaining--;
        }
      } else if (entry.isFile() && entry.name.endsWith(".jsonl")) {
        const info = await stat(path);
        if (info.mtimeMs < cutoff) {
          await rm(path);
          removed++;
          remaining--;
        }
      }
    }
    return remaining === 0;
  }

  await walk(root);
  return removed;
}
